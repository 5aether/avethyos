cmake_minimum_required(VERSION 3.20)
project(AvethyDesktopTheme LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SKIP_RPATH ON)
set(QT_MAJOR 6 CACHE STRING "Qt major version (5 or 6)")
find_package(Qt${QT_MAJOR} REQUIRED COMPONENTS Widgets)
add_library(avethy-palette STATIC palette.cpp palette.h)
target_link_libraries(avethy-palette PUBLIC Qt${QT_MAJOR}::Widgets)
add_library(avethy-style MODULE style.cpp style.h styleplugin.cpp)
target_link_libraries(avethy-style PRIVATE avethy-palette)
set_target_properties(avethy-style PROPERTIES
    OUTPUT_NAME avethy-style-qt${QT_MAJOR} LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/styles")
add_library(avethy-theme MODULE themeplugin.cpp)
target_link_libraries(avethy-theme PRIVATE avethy-palette)
set_target_properties(avethy-theme PROPERTIES
    OUTPUT_NAME avethy-theme-qt${QT_MAJOR} LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/generic")
include(CTest)
if(BUILD_TESTING)
add_executable(avethy-theme-check ../../tests/qt-theme-check.cpp)
target_link_libraries(avethy-theme-check PRIVATE Qt${QT_MAJOR}::Widgets)
add_library(qt-application-probe MODULE ../../tests/qt-application-probe.cpp)
target_link_libraries(qt-application-probe PRIVATE Qt${QT_MAJOR}::Widgets)
set_target_properties(qt-application-probe PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/generic")
enable_testing()
add_test(NAME native-theme COMMAND avethy-theme-check)
set_tests_properties(native-theme PROPERTIES ENVIRONMENT
    "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=;QT_PLUGIN_PATH=${CMAKE_BINARY_DIR}/plugins;QT_STYLE_OVERRIDE=Avethy;QT_QPA_GENERIC_PLUGINS=avethy-theme")
find_package(Qt${QT_MAJOR} QUIET COMPONENTS Quick)
if(TARGET Qt${QT_MAJOR}::Quick)
    add_executable(quickshell-controls-check ../../tests/qtquick-controls-check.cpp)
    target_link_libraries(quickshell-controls-check PRIVATE Qt${QT_MAJOR}::Quick Qt${QT_MAJOR}::Widgets)
    target_compile_definitions(quickshell-controls-check PRIVATE
        QUICKSHELL_QUICK_TEST="${CMAKE_CURRENT_SOURCE_DIR}/../../tests/qtquick-controls-check.qml")
    foreach(application IN ITEMS gui widgets)
        foreach(import IN ITEMS unversioned versioned)
            if(QT_MAJOR EQUAL 5 AND import STREQUAL "unversioned")
                continue() # Qt 5 requires an explicit Controls import version.
            endif()
            set(arguments)
            if(application STREQUAL "widgets")
                list(APPEND arguments --widgets)
            endif()
            if(import STREQUAL "versioned")
                list(APPEND arguments --versioned)
            endif()
            set(test_name "quick-${application}-${import}")
            add_test(NAME ${test_name} COMMAND quickshell-controls-check ${arguments})
            set_tests_properties(${test_name} PROPERTIES ENVIRONMENT
                "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=;QT_PLUGIN_PATH=${CMAKE_BINARY_DIR}/plugins;QT_STYLE_OVERRIDE=Avethy;QT_QPA_GENERIC_PLUGINS=avethy-theme;QT_QUICK_CONTROLS_STYLE=Avethy;QT_QUICK_CONTROLS_FALLBACK_STYLE=Fusion;QT_QUICK_CONTROLS_STYLE_PATH=${CMAKE_CURRENT_SOURCE_DIR}/qml;QML_IMPORT_PATH=${CMAKE_CURRENT_SOURCE_DIR}/qml")
        endforeach()
    endforeach()
    add_test(NAME quick-legacy-environment COMMAND quickshell-controls-check --widgets --versioned)
    set_tests_properties(quick-legacy-environment PROPERTIES ENVIRONMENT
        "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=;QT_PLUGIN_PATH=${CMAKE_BINARY_DIR}/plugins;QT_STYLE_OVERRIDE=Quickshell;QT_QPA_GENERIC_PLUGINS=avethy-theme;QT_QUICK_CONTROLS_STYLE=Quickshell;QT_QUICK_CONTROLS_FALLBACK_STYLE=Fusion;QT_QUICK_CONTROLS_STYLE_PATH=${CMAKE_CURRENT_SOURCE_DIR}/qml;QML_IMPORT_PATH=${CMAKE_CURRENT_SOURCE_DIR}/qml")
    add_test(NAME quick-legacy-desktop-environment COMMAND quickshell-controls-check --widgets --versioned)
    set_tests_properties(quick-legacy-desktop-environment PROPERTIES ENVIRONMENT
        "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=;QT_PLUGIN_PATH=${CMAKE_BINARY_DIR}/plugins;QT_STYLE_OVERRIDE=QuickshellDesktop;QT_QPA_GENERIC_PLUGINS=quickshell-theme;QT_QUICK_CONTROLS_STYLE=QuickshellDesktop;QT_QUICK_CONTROLS_FALLBACK_STYLE=Fusion;QT_QUICK_CONTROLS_STYLE_PATH=${CMAKE_CURRENT_SOURCE_DIR}/qml;QML_IMPORT_PATH=${CMAKE_CURRENT_SOURCE_DIR}/qml")
    if(QT_MAJOR EQUAL 5)
        add_test(NAME quick-legacy-missing-style-path COMMAND quickshell-controls-check --widgets --versioned)
        set_tests_properties(quick-legacy-missing-style-path PROPERTIES ENVIRONMENT
            "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=;QT_PLUGIN_PATH=${CMAKE_BINARY_DIR}/plugins;QT_STYLE_OVERRIDE=QuickshellDesktop;QT_QPA_GENERIC_PLUGINS=quickshell-theme;QT_QUICK_CONTROLS_STYLE=QuickshellDesktop;QT_QUICK_CONTROLS_FALLBACK_STYLE=Fusion;QT_QUICK_CONTROLS_STYLE_PATH=;QML_IMPORT_PATH=${CMAKE_CURRENT_SOURCE_DIR}/qml")
    endif()
endif()
endif()
